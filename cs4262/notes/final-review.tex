\section{Linear Regression}
\begin{itemize}
	\item MSE

	      \[
		      \min_{f} \; \frac{1}{N} \sum_{i=1}^{N} \Big(f\big(x^{(i)}\big) - y^{(i)}\Big)^2
	      \]
	\item Likelihood

	      \[
		      p(y \mid x, f, \sigma)
		      \;=\;
		      \prod_{i=1}^N p\!\big(y^{(i)} \mid x^{(i)}, f, \sigma\big).
	      \]
	\item Optimization function
	      \[
		      \min_{w} \; \frac{1}{N} \|y - Xw\|^2.
	      \]
	\item Closed form solution
	      \[
		      w^* = (X^T X)^{-1} X^T y
	      \]
	\item Ridge regression closed form solution
	      \[
		      w^* = (X^T X + \lambda I)^{-1} X^T y
	      \]

	\item Gradient decent formula
	      \[
		      w^{(t)} = w^{(t-1)} - \epsilon \nabla_w \mathcal{L}(w^{(t-1)})
	      \]
	\item Robust regression
	      \[
		      \mathcal{L}(w) = \sum_{i=1}^N \alpha^{(i)} \bigl\lVert y^{(i)} - f(x^{(i)}) \bigr\rVert^2
	      \]

	      \[
		      \alpha^{(i)} = h(\epsilon^{(i)})
		      = \frac{1}{\max(\delta, \epsilon)},
	      \]

	      $\delta$ = Threshold \\
	      $\epsilon$ = Data point error

\end{itemize}

\section{Support Vector Machines}
\begin{itemize}
	\item (Primal) model
	      \[
		      f(x) = \text{sign}(w^T x + b)
	      \]
	\item (Dual) model
	      \[
		      f(x) = \text{sign}\!\left( \sum_i \alpha^{(i)} y^{(i)} \langle x^{(i)}, x \rangle + b \right)
	      \]
	\item Hyperplane

	      \[
		      w^T x + b = 0
	      \]
	\item Margin
	      \[
		      \rho = \frac{2}{\|w\|}
	      \]
	\item (Hard Margin) optimization function
	      \[
		      \min_{w,b} \; w^Tw \; \; \text{s.t.} \; \; y^{(i)} \big(w^T x^{(i)} + b\big) \;\geq\; 1,
		      \; \; \forall i
	      \]

	\item (Soft Margin) optimization function

	      \[
		      \min_{w,b,\{\xi_i\}} \quad w^T w + C \sum_i \xi^{(i)} \; \; \text{s.t} \quad y^{(i)} \big(w^T x^{(i)} + b\big) \;\geq\; 1 - \xi^{(i)}, \; \; \forall i \;\; \xi^{(i)} \;\geq\; 0, \quad \forall i
	      \]
	\item Hinge loss
	      \[
		      \xi^{(i)} = \max \big(0, \, 1 - (w^T x^{(i)} + b)y^{(i)}\big)
	      \]
	\item Lagrangian
	      \[
		      \mathcal{L}(x, \alpha, \beta) = f(x) + \alpha g(x) + \beta h(x)
	      \]
	\item KKT conditions
	      \begin{enumerate}
		      \item Stationary:
		            \[
			            \nabla_x \mathcal{L}(x^*, \alpha^*, \beta^*) = 0
		            \]
		      \item Feasibility:
		            \[
			            \alpha^* \geq 0, \quad g(x^*) \leq 0, \quad h(x^*) = 0
		            \]
		      \item Complementary slackness:
		            \[
			            \alpha^* g(x^*) = 0
		            \]
	      \end{enumerate}
	\item SVM primal problem

	      \[
		      \min_{w,b} \;\; \tfrac{1}{2} w^T w
		      \quad \text{s.t.} \quad
		      y^{(i)} \big(w^T x^{(i)} + b \big) \geq 1
	      \]
	\item SVM dual problem
	      \[
		      \max_{\{\alpha^{(i)} \geq 0\}_i}
		      \;\; \min_{w,b} \left(
		      \tfrac{1}{2} w^T w \;+\;
		      \sum_i \alpha^{(i)} \Big[1 - y^{(i)} (w^T x^{(i)} + b) \Big]
		      \right)
	      \]

	\item (Primal) non-linear SVM formula
	      \[
		      f(x) = \text{sign}(w^T \phi (x) + b)
	      \]
	\item (Dual) non-linear SVM formula
	      \[
		      f(x) = \text{sign}\left( \sum_i \alpha^{(i)} y^{(i)} (\phi{(x)^T\phi{(x^{(i)})}}) + b \right)
	      \]

	\item Kernel trick

	      \[
		      K(x,z) = \langle \phi(x), \phi(z)\rangle.
	      \]

	\item (Primal) SVM regression model

	      \[
		      f(x) = w^T \phi(x) + b
	      \]
	\item (Dual) SVM regression model
	      \[
		      f(x) = \sum_i (\alpha^{(i)} - \beta^{(i)}) K(x^{(i)}, x) + b
	      \]
	      where
	      \[
		      K(x, x') = \phi(x)^T \phi(x')
	      \]

\end{itemize}

\section{Metrics}
\begin{itemize}
	\item Accuracy
	      \[
		      \frac{TP + TN}{TP + TN + FP + FN}
	      \]
	\item Precision
	      \[
		      \frac{TP}{TP+FP}
	      \]
	\item Recall
	      \[
		      \frac{TP}{TP + FN}
	      \]

	\item F1-Score
	      \[
		      F_1 = 2\frac{Precision \cdot Recall}{Precision + Recall}
	      \]
	\item True positive rate (TPR)
	      \[
		      \frac{TP}{TP+FN}
	      \]

	\item False positive rate (FPR)
	      \[
		      \frac{FP}{FP+TN}
	      \]
\end{itemize}

	      \[
		      X_K = \{ x^{(\pi(k))} \}_{k=1}^K

	      \[
		      d(x, x^{(i)}) < d(x, x^{(j)}) \quad \forall x^{(i)} \in X_K, \ \forall x^{(j)} \notin X_K

\section{Dimension Reduction}
\begin{itemize}
	\subsection{PCA}
	\item Reduction formulation

	      \[
		      f: \mathbb{R}^d \rightarrow \mathbb{R}^k \quad k<d
	      \]
	      \[
		      x \in \mathbb{R}^d
	      \]
	      \[
		      z = f(x) \in \mathbb{R}^k
	      \]
	      \[
		      \hat{x} = w w^T x
	      \]
	\item Mean of projection
	      \[
		      \mathbf{w}^T \left( \frac{1}{N} \sum_{i=1}^{N} \mathbf{x}_i \right)
		      = \mathbf{w}^T \bar{\mathbf{x}}
	      \]
	\item Variance of projection
	      \[
		      = w^T \left( \frac{1}{N} \sum_{i=1}^{N} (x_i - \bar{x})(x_i - \bar{x})^T \right) w
		      = w^T S w
	      \]
	\item Optimization problem
	      \[
		      \max_{w} \quad w^T S w \quad \text{s.t.} \quad w^T w = 1
	      \]
	\item Transpose trick

	      \[
		      S = \frac{1}{N}XX^T \in \mathbb{R}^{DXD}
	      \]

	      If d $>>$ n we can instead use this:
	      \[
		      G = \frac{1}{N}X^TX \in R^{NXN}
	      \]
\end{itemize}
\subsection{PCA}
\begin{itemize}
	\item Distance between class means
	      \[
		      \begin{aligned}
			       & w^T \left( \frac{1}{K} \sum_{k} (\mu^{(k)} - \bar{\mu})(\mu^{(k)} - \bar{\mu})^T \right) w \\
			       & = w^T S_b w
		      \end{aligned}
	      \]
	\item Within class variance

			       & = w^T S_w w
		      \end{aligned}
	      \]

\end{itemize}

\section{K-means clustering}
\begin{itemize}
	\item Optimization problem
	      \[
		      \min_{C, Z} \quad \sum_{k=1}^{K} \sum_{i \,:\, c(i)=k} \left\| x^{(i)} - \mu^{(k)} \right\|^2
	      \]
	\item Optimization subproblems
	      \begin{itemize}
		      \item \textbf{For fixed } $Z = [\mu^{(1)}, \dots, \mu^{(K)}]^T$, \textbf{optimize } $C$:
		            \[
			            \min_{C} \quad \sum_{i=1}^{N} \left\| x^{(i)} - \mu^{(c(i))} \right\|^2
		            \]

		      \item \textbf{For fixed } $C$, \textbf{optimize } $Z$:
		            \[
			            \min_{Z} \quad \sum_{k=1}^{K} \sum_{i} \left\| x^{(i)} - \mu^{(k)} \right\|^2\quad s.t \; c^{(i)} = k
		            \]

	      \end{itemize}

	\item Process
	      \begin{enumerate}
		      \item Initilaize with K random centroids
		      \item Repeat until convergence
		            \begin{itemize}
			            \item Assign a cluster to every sample $x^{(i)} \Rightarrow c^{(i)} = \arg\min_k dist(x^{(i)}, u^{(k)})$
			            \item Update centroids according to clusters: \[
				                  \mu^{(k)} =
				                  \frac{\displaystyle \sum_{i=1}^{N} \mathbf{1}_{c_i = k} \, x^{(i)}}
				                  {\displaystyle \sum_{i=1}^{N} \mathbf{1}_{c_i = k}}
			                  \]

		            \end{itemize}
	      \end{enumerate}
	\item The silhouette  method
	      \[
		      s(o) = \frac{b(o) - a(o)}{\max\{a(o),\, b(o)\}}
	      \]

\end{itemize}

\section{Gaussian Mixture Models}
\begin{itemize}
	\item Probability to observe a value given normal distribution
	      \[
		      p(x; \mu, \sigma) = \mathcal{N}(x; \mu, \sigma) = \frac{1}{\sqrt{2\pi\sigma^2}} \, e^{ -\frac{(x - \mu)^2}{2\sigma^2} }
	      \]

	\item Maximum Likelihood Estimation (MLE)
	      \[
		      \arg\max_{\mu, \sigma} \, p(x_1, \dots, x_N; \mu, \sigma) = \prod_{n=1}^{N} p(x_n; \mu, \sigma)
	      \]

	\item Multiple gaussians probability to observe a value given normal distribution
	      \[
		      p(x; \{(\alpha_k, \mu_k, \sigma_k)\}_{k=1}^K)
		      = \sum_{k=1}^{K} \alpha_k \, \mathcal{N}(x; \mu_k, \sigma_k)
		      = \sum_{k=1}^{K} \frac{\alpha_k}{\sqrt{2\pi\sigma_k^2}}
		      \exp\left( -\frac{(x - \mu_k)^2}{2\sigma_k^2} \right)
	      \]

	\item MoG likelihood of the full dataset
	      \[
		      \prod_{n=1}^{N} \sum_{k=1}^{K} \alpha_k \, \mathcal{N}(x_n; \mu_k, \sigma_k)
	      \]
	\item EM algorithm
	      \begin{enumerate}
		      \item E-step (Soft assignments)
		            \[
			            r_n^k = p(z_k = 1 \mid x_n)
			            = \frac{ \alpha_k \, \mathcal{N}(x_n; \mu_k, \sigma_k^2) }
			            { \sum\limits_{i=1}^{K} \alpha_i \, \mathcal{N}(x_n; \mu_i, \sigma_i^2) }
		            \]
		      \item M-step (Update parameters)

		            \medskip
		            1. Update Mixture Coefficients

		            \[
			            \alpha_k = \frac{N_k}{N}
			            \; \; \text{for} \; \;
			            N_k = \sum_{n=1}^{N} r_n^k
		            \]

		            2. Update Means

		            \[
			            \mu_k = \frac{1}{N_k} \sum_{n=1}^{N} r_n^k x_n
		            \]

		            3. Update Covariances

		            \[
			            \Sigma_k = \frac{1}{N_k} \sum_{n=1}^{N} r_n^k (x_n - \mu_k)(x_n - \mu_k)^T
		            \]
	      \end{enumerate}

\end{itemize}

\subsection{Spectral clustering}
\begin{itemize}
	\item Degree of nodes
	      \[
		      d_i = \sum_j w_{i,j}
	      \]
	      The degree of node \(i\) is the sum of weights to connected nodes.

	\item Volume of a set
	      \[
		      \text{vol}(A) = \sum_{i \in A} d_i, \quad A \subseteq V
	      \]
	      The volume of a set \(A\) is the sum of node degrees over all nodes in set \(A\).
	\item Graph cut
	      \[
		      \text{cut}(A, B) = \sum_{i \in A,\, j \in B} w_{ij}
	      \]
	\item Normalized Cut
	      \[
		      \text{Ncut}(A, B) = \text{cut}(A, B) \cdot \frac{\text{Vol}(A) + \text{Vol}(B)}{\text{Vol}(A)\text{Vol}(B)}
	      \]
	\item Laplacian
	      \[
		      L = D - W
	      \]
	\item Relaxed optimization problem

	      \[
		      \min_{\mathbf{f}} \quad \mathbf{f}^T L \mathbf{f} \quad \text{subject to} \quad \mathbf{f}^T D \mathbf{f} = 1
	      \]

	\item Closed form solution

	      \[
		      D^{-1} L \mathbf{f} = \lambda \mathbf{f}
	      \]

	\item Algorithm

	      \begin{enumerate}
		      \item Compute the Laplacian:
		            \[
			            L = D - W
		            \]

		      \item Solve the generalized eigenvalue problem:
		            \[
			            D^{-1}L \, \mathbf{f}_k = \lambda_k \mathbf{f}_k
		            \]
		            and compute the first \( k \) eigenvectors \( \mathbf{f}_1, \dots, \mathbf{f}_k \)

		      \item Let \( U \in \mathbb{R}^{N \times k} \) be the matrix containing \( \mathbf{f}_1, \dots, \mathbf{f}_k \) as columns

		      \item For \( i = 1, \dots, N \), let \( \mathbf{y}_i \in \mathbb{R}^k \) be the vector corresponding to the \( i \)-th row of \( U \)

		      \item Cluster the points \( \{ \mathbf{y}_i \}_{i=1}^N \subset \mathbb{R}^k \) using the \( k \)-means algorithm into clusters \( C_1, \dots, C_k \)
	      \end{enumerate}
\end{itemize}

\section{Generative Adversarial Networks}
\begin{itemize}
	\item Objective problem
	      \[
		      \min_{\theta_\psi} \max_{\theta_\eta}
		      \left[
			      \frac{1}{N} \sum_i \log \big( \eta(x_i; \theta_\eta) \big)
			      +
			      \frac{1}{M} \sum_j \log \big( 1 - \eta(\hat{x}_j; \theta_\eta) \big)
			      \right],
	      \]
	\item Algorithm

	      \begin{enumerate}
		      \item \textbf{For each training iteration:}
		            \begin{enumerate}
			            \item \textbf{Discriminator updates (repeated $k$ times):}
			                  \begin{itemize}
				                  \item Sample a minibatch of $m$ noise samples $\{ z^{(1)}, \ldots, z^{(m)} \}$ from noise prior $p_g(z)$.
				                  \item Sample a minibatch of $m$ real examples $\{ x^{(1)}, \ldots, x^{(m)} \}$ from data distribution $p_{\text{data}}(x)$.
				                  \item Update the discriminator by ascending its stochastic gradient:
				                        \[
					                        \nabla_{\theta_d} \frac{1}{m} \sum_{i=1}^{m}
					                        \Big[
						                        \log D(x^{(i)}) + \log \big(1 - D(G(z^{(i)}))\big)
						                        \Big].
				                        \]
			                  \end{itemize}

			            \item \textbf{Generator update:}
			                  \begin{itemize}
				                  \item Sample a minibatch of $m$ noise samples $\{ z^{(1)}, \ldots, z^{(m)} \}$ from noise prior $p_g(z)$.
				                  \item Update the generator by descending its stochastic gradient:
				                        \[
					                        \nabla_{\theta_g} \frac{1}{m} \sum_{i=1}^{m}
					                        \log \big(1 - D(G(z^{(i)}))\big).
				                        \]
			                  \end{itemize}
		            \end{enumerate}
	      \end{enumerate}
\end{itemize}

\section{Deep sets}
\begin{itemize}
	\item Algorithm
	      \begin{enumerate}
		      \item \textbf{Generate all permutations:} \\
		            Since the order of elements shouldn’t matter, you can imagine training the neural network to produce the same output for all possible input orders.
		            \begin{itemize}
			            \item For \( N = 5 \), that’s \( 5! = 120 \) permutations — quickly intractable as \( N \) grows.
		            \end{itemize}

		      \item \textbf{Neural network \(\psi\):} \\
		            Each permutation of the set is fed into a neural network \(\psi\), producing embeddings \( h_1, h_2, h_3, \ldots \)

		      \item \textbf{Aggregate function \(f\):} \\
		            These embeddings are combined using a \textbf{permutation-invariant} aggregation (e.g., sum, mean, max).

		      \item \textbf{Final mapping \(\phi\):} \\
		            The aggregated representation is passed through another neural network \(\phi\) to produce the final output \( y \).
	      \end{enumerate}
\end{itemize}

\section{Reinforcement Learning}
\begin{itemize}
	\item Markov Property
	      \[
		      P(S_{t+1} = s_{t+1} \mid
		      S_t = s_t, A_t = a_t,
		      S_{t-1} = s_{t-1}, A_{t-1} = a_{t-1},
		      \ldots,
		      S_0 = s_0)
		      =
		      P(S_{t+1} = s_{t+1} \mid S_t = s_t, A_t = a_t).
	      \]
	\item Markov objective
	      \[
		      \sum_{t=0}^{\infty} \gamma^t r_t.
	      \]
	\item Optimal policy
	      \[
		      \pi^{*} = \arg\max_{\pi} \mathbb{E}\!\left[ \sum_{t \ge 0} \gamma^{t} r_{t} \,\middle|\, \pi \right]
		      \quad \text{with} \quad
		      s_{0} \sim p(s_{0}),\;
		      a_{t} \sim \pi(\cdot \mid s_{t}),\;
		      s_{t+1} \sim p(\cdot \mid s_{t}, a_{t}).
	      \]
	\item Value function
	      \[
		      V^{\pi}(s) =
		      \mathbb{E} \!\left[ \sum_{t \ge 0} \gamma^{t} r_t \,\middle|\, s_0 = s,\, \pi \right].
	      \]

	\item Q-value function
	      \[
		      Q^{\pi}(s, a) =
		      \mathbb{E} \!\left[ \sum_{t \ge 0} \gamma^{t} r_t \,\middle|\, s_0 = s,\; a_0 = a,\; \pi \right].
	      \]

	\item Bellman Equation
	      \[
		      Q^{*}(s, a)
		      = \max_{\pi}
		      \mathbb{E} \!\left[
			      \sum_{t \ge 0} \gamma^{t} r_t \,\middle|\,
			      s_0 = s,\; a_0 = a,\; \pi
			      \right].
	      \]

	      \(Q^{*}\) satisfies the following:

	      \[
		      Q^{*}(s, a)
		      = \mathbb{E}_{s' \sim \mathcal{E}}
		      \left[
			      r + \gamma \max_{a'} Q^{*}(s', a')
			      \,\middle|\, s, a
			      \right].
	      \]

	\item Recovering optimal policy

	      \[
		      \pi^{*}(s) = \arg\max_{a} Q^{*}(s,a).
	      \]
	      aka for each state pick the action with the highest Q-value

	\item Q-learning algorithm

	      Initialize $Q(s,a)$, $\forall s \in \mathcal{S}, a \in \mathcal{A}(s)$, arbitrarily, and
	      $Q(\text{terminal-state}, \cdot) = 0$.

	      Repeat (for each episode):

	      \quad Initialize $S$

	      \quad Repeat (for each step of episode):

	      \qquad Choose $A$ from $S$ using policy derived from $Q$ (e.g., $\varepsilon$-greedy)

	      \qquad Take action $A$, observe $R$, $S'$

	      \qquad $Q(S,A) \leftarrow Q(S,A) + \alpha\left[ R + \gamma \max_a Q(S',a) - Q(S,A) \right]$

	      \qquad $S \leftarrow S'$

	      \quad until $S$ is terminal.


	      \[
		      Q_{i+1}(s,a)
		      = \mathbb{E}\left[\, r + \gamma \max_{a'} Q_i(s', a') \mid s, a \,\right]
	      \]

	      \[
		      Q_i \;\text{will converge to}\; Q^{*} \;\text{as}\; i \to \infty
	      \]

	\item Deep Q-learning approximation
	      \[
		      Q(s,a;\theta) \approx Q^{*}(s,a),
	      \]
	\item Deep Q-learning target
	      The Bellman target for iteration $i$:
	      \[
		      y_i = r + \gamma \max_{a'} Q(s',a';\theta_{i-1}).
	      \]
	\item Deep Q-learning loss function
	      \[
		      L_i(\theta_i) = \mathbb{E}\Big[(\, y_i - Q(s,a;\theta_i) \,)^2\Big].
	      \]

\end{itemize}
